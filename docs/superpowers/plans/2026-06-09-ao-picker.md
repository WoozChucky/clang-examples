```cpp
#pragma once

#include <nlohmann/json.hpp>

// Resolve the persisted AO mode (0=Off, 1=SSAO, 2=HBAO, 3=GTAO) from the settings "ssao" JSON
// object, migrating the legacy boolean "enabled" key.
//
// Precedence:
//   1. "mode" (int, in range [0,3])  -> use it
//   2. legacy "enabled" (bool)        -> true => SSAO(1), false => Off(0)
//   3. neither / out-of-range mode    -> default SSAO(1)
//
// Int-based + dependency-free (no engine enum) so it is unit-testable from the common settings
// layer; the Engine maps this int to/from its AoMode enum at the edges.
inline int ResolveAoMode(const nlohmann::json& ssao)
{
    if (ssao.contains("mode") && ssao["mode"].is_number_integer()) {
        const int m = ssao["mode"].get<int>();
        if (m >= 0 && m <= 3) return m;
    }
    if (ssao.contains("enabled") && ssao["enabled"].is_boolean()) {
        return ssao["enabled"].get<bool>() ? 1 : 0;
    }
    return 1; // default SSAO
}
```

- [ ] **Step 2: Add the `HBAO_CS` shader string** — in `SsaoRenderPass.cpp`, inside the anonymous namespace after `AO_CS`, add. Bindings IDENTICAL to `AO_CS` (b0/t1/t2/s3/u4) so it reuses `m_AoLayout`:

```cpp
static const char* HBAO_CS = R"(
Texture2D       uNormal   : register(t1);
Texture2D       uWorldPos : register(t2);
SamplerState    uPt       : register(s3);
RWTexture2D<float> uOut    : register(u4);
cbuffer SsaoCB : register(b0) {
    float4x4 uViewProj; float4x4 uView; float4 uKernel[16]; float4 uParams; float4 uRtSize;
};
#define DIRS 4
#define STEPS 8
#define PI 3.14159265
[numthreads(8,8,1)]
void main_cs(uint3 tid : SV_DispatchThreadID) {
    uint2 px = tid.xy;
    if (px.x >= (uint)uRtSize.x || px.y >= (uint)uRtSize.y) return;
    float2 uv = (float2(px) + 0.5) * uRtSize.zw;
    float3 Nw = uNormal.SampleLevel(uPt, uv, 0).xyz;
    if (dot(Nw,Nw) < 0.5) { uOut[px] = 1.0; return; }
    Nw = normalize(Nw);
    float3 Pw = uWorldPos.SampleLevel(uPt, uv, 0).xyz;
    float3 Pv = mul(uView, float4(Pw,1)).xyz;            // view-space pos (RH, z negative)
    float3 Nv = normalize(mul((float3x3)uView, Nw));     // view-space normal
    float radius = uParams.x, intensity = uParams.z, power = uParams.w;
    // World-space tangent basis (same idea as AO_CS), march directions around it.
    float3 up = abs(Nw.y) < 0.99 ? float3(0,1,0) : float3(1,0,0);
    float3 T0 = normalize(cross(up, Nw));
    float3 B0 = cross(Nw, T0);
    // per-pixel rotation: SAME 4x4 tile as AO_CS so the 4x4 blur cancels it.
    float2 fpx = floor(uv * uRtSize.xy); float2 cell = fmod(fpx, 4.0);
    float jit = frac(sin((cell.y*4.0 + cell.x) * 12.9898) * 43758.5453);
    float occ = 0.0;
    [loop] for (int d = 0; d < DIRS; ++d) {
        float ang = (float(d) + jit) * (PI / DIRS);
        float3 dir = T0 * cos(ang) + B0 * sin(ang);      // world-space march dir in the tangent plane
        float maxSin = 0.0;                               // tangent baseline (horizon sine), >=0
        [loop] for (int s = 1; s <= STEPS; ++s) {
            float3 sw = Pw + dir * (radius * (float(s) / STEPS));
            float4 clip = mul(uViewProj, float4(sw,1));
            if (clip.w <= 1e-5) continue;
            float2 suv = clip.xy/clip.w * 0.5 + 0.5; suv.y = 1 - suv.y;
            if (any(suv < 0) || any(suv > 1)) continue;
            float3 Ow = uWorldPos.SampleLevel(uPt, suv, 0).xyz;
            float3 Ov = mul(uView, float4(Ow,1)).xyz;     // occluder view pos
            float3 h  = Ov - Pv;                          // horizon vector in view space
            float  len = length(h);
            if (len < 1e-4 || len > radius) continue;
            float sinH = dot(normalize(h), Nv);           // elevation above the surface plane
            float w = saturate(1.0 - len / radius);       // range falloff
            maxSin = max(maxSin, sinH * w);
        }
        occ += maxSin;
    }
    occ = (occ / DIRS) * intensity;
    uOut[px] = pow(saturate(1.0 - occ), power);
}
)";
```

- [ ] **Step 2: Add the `GTAO_CS` shader string** — same bindings/CB/`#define`s as HBAO. Per direction, find the horizon on BOTH sides of the view vector and integrate the cosine-weighted arc relative to the projected normal:

```cpp
static const char* GTAO_CS = R"(
Texture2D       uNormal   : register(t1);
Texture2D       uWorldPos : register(t2);
SamplerState    uPt       : register(s3);
RWTexture2D<float> uOut    : register(u4);
cbuffer SsaoCB : register(b0) {
    float4x4 uViewProj; float4x4 uView; float4 uKernel[16]; float4 uParams; float4 uRtSize;
};
#define DIRS 4
#define STEPS 8
#define PI 3.14159265
// March one direction, return the horizon angle (radians from the view vector) on that side.
float horizonAngle(float3 Pw, float3 Pv, float3 dir, float radius, float4x4 vp, float4x4 vw,
                   Texture2D wp, SamplerState pt) {
    float cosH = -1.0; // cos of max horizon angle found (start at the widest)
    [loop] for (int s = 1; s <= STEPS; ++s) {
        float3 sw = Pw + dir * (radius * (float(s)/STEPS));
        float4 clip = mul(vp, float4(sw,1));
        if (clip.w <= 1e-5) continue;
        float2 suv = clip.xy/clip.w*0.5+0.5; suv.y = 1-suv.y;
        if (any(suv<0)||any(suv>1)) continue;
        float3 Ov = mul(vw, float4(wp.SampleLevel(pt,suv,0).xyz,1)).xyz;
        float3 h = Ov - Pv; float len = length(h);
        if (len < 1e-4 || len > radius) continue;
        float c = dot(normalize(h), normalize(-Pv)); // angle vs view direction
        cosH = max(cosH, c);
    }
    return acos(clamp(cosH, -1.0, 1.0));
}
[numthreads(8,8,1)]
void main_cs(uint3 tid : SV_DispatchThreadID) {
    uint2 px = tid.xy;
    if (px.x >= (uint)uRtSize.x || px.y >= (uint)uRtSize.y) return;
    float2 uv = (float2(px)+0.5)*uRtSize.zw;
    float3 Nw = uNormal.SampleLevel(uPt, uv, 0).xyz;
    if (dot(Nw,Nw) < 0.5) { uOut[px] = 1.0; return; }
    Nw = normalize(Nw);
    float3 Pw = uWorldPos.SampleLevel(uPt, uv, 0).xyz;
    float3 Pv = mul(uView, float4(Pw,1)).xyz;
    float3 Nv = normalize(mul((float3x3)uView, Nw));
    float radius = uParams.x, intensity = uParams.z, power = uParams.w;
    float3 up = abs(Nw.y) < 0.99 ? float3(0,1,0) : float3(1,0,0);
    float3 T0 = normalize(cross(up, Nw)); float3 B0 = cross(Nw, T0);
    float2 fpx = floor(uv*uRtSize.xy); float2 cell = fmod(fpx,4.0);
    float jit = frac(sin((cell.y*4.0+cell.x)*12.9898)*43758.5453);
    float vis = 0.0;
    [loop] for (int d = 0; d < DIRS; ++d) {
        float ang = (float(d)+jit)*(PI/DIRS);
        float3 dir = T0*cos(ang) + B0*sin(ang);
        float h1 = horizonAngle(Pw, Pv,  dir, radius, uViewProj, uView, uWorldPos, uPt);
        float h2 = horizonAngle(Pw, Pv, -dir, radius, uViewProj, uView, uWorldPos, uPt);
        // cosine-weighted visible arc between the two horizons (GTAO inner integral, simplified)
        vis += (sin(h1) + sin(h2)) * 0.5;
    }
    float ao = vis / DIRS;            // ~1 unoccluded, ->0 occluded
    ao = pow(saturate(ao), power);
    ao = lerp(1.0, ao, saturate(intensity));
    uOut[px] = ao;
}
)";
```

## Notes / gotchas

- **Shader bugs surface only at GUI smoke** (engine history: SSAO-compute hit 4 such gotchas). T3/T4 each end with a human smoke + likely a tuning/sign-fix iteration — budget for it; don't declare a shader "done" on build-green alone.
- **Binding parity:** HBAO/GTAO MUST bind exactly `b0/t1/t2/s3/u4` (same as `AO_CS`) to reuse `m_AoLayout`. Any change = flat-binding validation error / AV.
- **CS texture sampling:** use `.SampleLevel(s, uv, 0)` (NOT `.Sample`) — `.Sample` is illegal in CS < SM6.6 (the engine's recorded gotcha).
- **Named binding-set handle:** the AO dispatch already uses a named `nvrhi::BindingSetHandle bs` local (not an inline temporary) — keep it; an inline temporary dangles → AV (recorded gotcha).
- **View-space sign:** RH view-Z is negative; the HBAO/GTAO math derives view P/N via `uView` matching `AO_CS`. If AO looks inverted, the horizon/elevation sign is the first suspect.
- **Pipeline lifetime:** the new pipelines are lazily built in `Render` and dropped in `OnResize`/`Shutdown` exactly like `m_AoPipeline` — a backend swap/resize must not leave them bound to a dead device.
- **Breaking settings:** old `engine_settings.json` `renderer.ssao.enabled` is migrated by `ResolveAoMode`; no `mode` key → defaults to SSAO. Expected per early-dev policy.
